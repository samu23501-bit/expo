import React, { useState, useEffect } from "react";
import { View, Text, TextInput, TouchableOpacity, FlatList, StyleSheet, Dimensions } from "react-native";
import AsyncStorage from "@react-native-async-storage/async-storage";
import { PieChart } from "react-native-chart-kit";

const screenWidth = Dimensions.get("window").width;

export default function App() {
  const [assets, setAssets] = useState([]);
  const [name, setName] = useState("");
  const [value, setValue] = useState("");

  useEffect(() => {
    loadAssets();
  }, []);

  useEffect(() => {
    saveAssets();
  }, [assets]);

  const loadAssets = async () => {
    const stored = await AsyncStorage.getItem("assets");
    if (stored) setAssets(JSON.parse(stored));
  };

  const saveAssets = async () => {
    await AsyncStorage.setItem("assets", JSON.stringify(assets));
  };

  const total = assets.reduce((acc, asset) => acc + Number(asset.value), 0);

  const chartData = assets.map((asset, index) => ({
    name: asset.name,
    amount: asset.value,
    color: `hsl(${index * 60}, 70%, 50%)`,
    legendFontColor: "#333",
    legendFontSize: 14,
  }));

  const addAsset = () => {
    if (!name || !value) return;
    setAssets([...assets, { name, value: Number(value) }]);
    setName("");
    setValue("");
  };

  const removeAsset = (index) => {
    setAssets(assets.filter((_, i) => i !== index));
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>Mi Cartera</Text>

      <View style={styles.inputRow}>
        <TextInput
          placeholder="Nombre del activo"
          value={name}
          onChangeText={setName}
          style={styles.input}
        />
        <TextInput
          placeholder="Valor (€)"
          value={value}
          onChangeText={setValue}
          keyboardType="numeric"
          style={styles.input}
        />
      </View>

      <TouchableOpacity style={styles.addButton} onPress={addAsset}>
        <Text style={styles.addButtonText}>Añadir activo</Text>
      </TouchableOpacity>

      {assets.length > 0 && (
        <PieChart
          data={chartData}
          width={screenWidth - 20}
          height={220}
          chartConfig={{
            backgroundGradientFrom: "#fff",
            backgroundGradientTo: "#fff",
            decimalPlaces: 0,
            color: (opacity = 1) => `rgba(0, 0, 0, ${opacity})`,
          }}
          accessor="amount"
          backgroundColor="transparent"
          paddingLeft="10"
          absolute
        />
      )}

      <FlatList
        data={assets}
        keyExtractor={(_, index) => index.toString()}
        renderItem={({ item, index }) => (
          <View style={styles.assetRow}>
            <Text>
              {item.name} — {item.value}€ ({total ? ((item.value / total) * 100).toFixed(1) : 0}%)
            </Text>
            <TouchableOpacity onPress={() => removeAsset(index)}>
              <Text style={styles.delete}>Eliminar</Text>
            </TouchableOpacity>
          </View>
        )}
      />
    </View>
  );
}

const styles = StyleSheet.create({
  container: {
    flex: 1,
    backgroundColor: "#f2f2f2",
    padding: 16,
    paddingTop: 50,
  },
  title: {
    fontSize: 22,
    fontWeight: "bold",
    marginBottom: 16,
  },
  inputRow: {
    flexDirection: "row",
    gap: 8,
  },
  input: {
    flex: 1,
    backgroundColor: "#fff",
    padding: 10,
    borderRadius: 10,
  },
  addButton: {
    backgroundColor: "#111",
    padding: 12,
    borderRadius: 12,
    marginTop: 10,
    marginBottom: 20,
    alignItems: "center",
  },
  addButtonText: {
    color: "#fff",
    fontWeight: "bold",
  },
  assetRow: {
    flexDirection: "row",
    justifyContent: "space-between",
    backgroundColor: "#fff",
    padding: 12,
    borderRadius: 12,
    marginBottom: 8,
  },
  delete: {
    color: "red",
    fontWeight: "bold",
  },
});
